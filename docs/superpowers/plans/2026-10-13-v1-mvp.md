# Library Manager Reader — V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the V1 offline e-reader app (Android/iOS, Expo SDK 57) per the validated spec: EPUB/MOBI/AZW3/CBZ reading via foliate-js in a WebView, CBR converted at import, library with progress/TOC/bookmarks/typography, sync-ready SQLite schema.

**Architecture:** Three layers — UI (Expo Router screens), core (SQLite repos, import pipeline, bridge orchestration, i18n), and a rendering engine (foliate-js bundled with esbuild, loaded in a WebView, fed book bytes as base64 chunks over a versioned JSON postMessage protocol). Original book files are stored untouched in the app documents directory.

**Tech Stack:** Expo SDK 57 (React Native 0.86, React 19.2), TypeScript strict, Expo Router, expo-sqlite, expo-file-system, expo-document-picker, react-native-webview, foliate-js, fflate, node-unrar-js, zustand, i18next, Jest (+ jest-expo, @testing-library/react-native, node:sqlite for repo tests), Maestro (E2E smoke).

**Spec:** `docs/superpowers/specs/2026-10-13-library-manager-reader-design.md` (read it first — the plan argues from it).

## Global Constraints

- Expo SDK **57** stable, TypeScript **strict** (`"strict": true` in tsconfig).
- **No new native dependencies** beyond: expo-sqlite, expo-file-system, expo-document-picker, react-native-webview, expo-localization (all Expo-official). Pure-JS deps allowed (foliate-js, fflate, node-unrar-js, zustand, i18next, base64-arraybuffer, @fontsource/literata).
- Original book files are **never modified** after import (spec §6). CBR is converted to CBZ at import and the CBZ is the canonical file; `original_format` records provenance.
- Every persisted row carries UUID (v7) PK, `updated_at`; deletions are tombstones (`deleted_at`), purged after **30 days** (spec §5).
- **No hardcoded UI strings** — all copy through i18next (fr + en, key parity enforced by test).
- All UI copy visible to users is French/English via i18n; timestamps stored as epoch **milliseconds** (INTEGER).
- Engine↔app protocol is versioned (`PROTOCOL_VERSION = 1`); unknown incoming messages are ignored + logged (spec §7.4).
- Reader settings persisted globally in `settings` table; theme applies to UI and book content.
- Commits: conventional commits (`feat:`, `test:`, `chore:`), one commit per task minimum.

```ts
export type AppToEngine =

  | { type: 'dataEnd' }

  | { type: 'progress'; cfi: string | null; fraction: number; sectionIndex: number }
  | { type: 'bookmarkCaptured'; target: string; fraction: number; excerpt: string }
  | { type: 'extractResult'; requestId: string; ok: true; metadata: EngineMetadataPayload; cbzBase64: string | null }
```

`src/core/bridge/codec.ts`:

```ts
import { ENGINE_TO_APP_TYPES, type AppToEngine, type EngineToApp } from './protocol';

export function encodeMessage(m: AppToEngine | EngineToApp): string {
  return JSON.stringify(m);
}

export type ParsedEngineMessage = EngineToApp | { type: 'unknown'; raw: string };

export function parseMessage(raw: string): ParsedEngineMessage {
  let value: unknown;
  try {
    value = JSON.parse(raw);
  } catch {
    return { type: 'unknown', raw };
  }
  if (typeof value === 'object' && value !== null) {
    const t = (value as { type?: unknown }).type;
    if (typeof t === 'string' && (ENGINE_TO_APP_TYPES as readonly string[]).includes(t)) {
      return value as EngineToApp;
    }
  }
  return { type: 'unknown', raw };
}
```

`src/core/uuid.ts`:

```ts
import { getRandomValues } from 'expo-crypto';

export function uuidV7(random: (n: number) => Uint8Array = (n) => getRandomValues(new Uint8Array(n))): string {
  const b = random(16);
  const ts = Date.now();
  b[0] = (ts / 2 ** 40) & 0xff;
  b[1] = (ts / 2 ** 32) & 0xff;
  b[2] = (ts / 2 ** 24) & 0xff;
  b[3] = (ts / 2 ** 16) & 0xff;
  b[4] = (ts / 2 ** 8) & 0xff;
  b[5] = ts & 0xff;
  b[6] = ((b[6] ?? 0) & 0x0f) | 0x70;
  b[8] = ((b[8] ?? 0) & 0x3f) | 0x80;
  const hex = Array.from(b, (x) => x.toString(16).padStart(2, '0')).join('');
  return `${hex.slice(0, 8)}-${hex.slice(8, 12)}-${hex.slice(12, 16)}-${hex.slice(16, 20)}-${hex.slice(20)}`;
}
```

```ts
export interface DbDriver {
  exec(sql: string): Promise<void>;
  run(sql: string, params?: unknown[]): Promise<{ changes?: number }>;
  get<T>(sql: string, params?: unknown[]): Promise<T | null>;
```

`src/db/nodeSqliteDriver.ts`:

```ts
import type { DatabaseSync } from 'node:sqlite';
import type { DbDriver } from './driver';

export function createNodeSqliteDriver(db: DatabaseSync): DbDriver {
  return {
    async exec(sql) { db.exec(sql); },
    async run(sql, params = []) { const info = db.prepare(sql).run(...params); return { changes: Number(info.changes) }; },
    async get<T>(sql, params = []) { return (db.prepare(sql).get(...params) as T | undefined) ?? null; },
    async all<T>(sql, params = []) { return db.prepare(sql).all(...params) as T[]; },
  };
}
```

```ts
import { openDatabaseSync } from 'expo-sqlite';
import type { DbDriver } from './driver';

export function createExpoDriver(file = 'lmr.db'): DbDriver {
  const db = openDatabaseSync(file);
  return {
    async exec(sql) { db.execSync(sql); },
    async run(sql, params = []) { const r = db.runSync(sql, params); return { changes: r.changes }; },
    async get(sql, params = []) { return (db.getFirstSync(sql, params) as unknown) ?? null; },
    async all(sql, params = []) { return db.getAllSync(sql, params) as never; },
  };
}
```

`src/db/migrate.ts`:

```ts
import type { DbDriver } from './driver';

export const CURRENT_SCHEMA_VERSION = 1;

const MIGRATIONS: Record<number, string> = {
  1: `
    CREATE TABLE books (
      id TEXT PRIMARY KEY,
      title TEXT NOT NULL,
      author TEXT,
      format TEXT NOT NULL,
      original_format TEXT,
      file_name TEXT NOT NULL,
      file_size INTEGER NOT NULL,
      language TEXT,
      publisher TEXT,
      drm_identifier TEXT,
      added_at INTEGER NOT NULL,
      updated_at INTEGER NOT NULL,
      deleted_at INTEGER,
      last_read_at INTEGER
    );
    CREATE INDEX idx_books_deleted ON books(deleted_at);
    CREATE TABLE reading_progress (
      book_id TEXT PRIMARY KEY REFERENCES books(id) ON DELETE CASCADE,
      cfi TEXT,
      fraction REAL NOT NULL,
      section_index INTEGER,
      updated_at INTEGER NOT NULL
    );
    CREATE TABLE bookmarks (
      id TEXT PRIMARY KEY,
      book_id TEXT NOT NULL REFERENCES books(id) ON DELETE CASCADE,
      cfi TEXT NOT NULL,
      fraction REAL NOT NULL,
      preview_text TEXT,
      created_at INTEGER NOT NULL,
      updated_at INTEGER NOT NULL,
      deleted_at INTEGER
    );
    CREATE INDEX idx_bookmarks_book ON bookmarks(book_id, deleted_at);
    CREATE TABLE settings (
      key TEXT PRIMARY KEY,
      value TEXT NOT NULL,
      updated_at INTEGER NOT NULL
    );
  `,
};

export async function migrate(db: DbDriver): Promise<void> {
  await db.exec('PRAGMA foreign_keys = ON;');
  const row = await db.get<{ user_version: number }>('PRAGMA user_version');
  const current = row?.user_version ?? 0;
  for (let v = current + 1; v <= CURRENT_SCHEMA_VERSION; v++) {
    const sql = MIGRATIONS[v];
    if (!sql) throw new Error(`Missing migration for schema version ${v}`);
    await db.exec(sql);
    await db.exec(`PRAGMA user_version = ${v};`);
  }
}
```

```ts
export function createSettingsRepo(db: DbDriver) {

    const row = await db.get<{ value: string }>('SELECT value FROM settings WHERE key = ?', [key]);

    );
  };

    getUiPrefs: () => read(UI_KEY, DEFAULT_UI_PREFS),
    putUiPrefs: (p: UiPrefs) => write(UI_KEY, p),
```

```ts
import type { DbDriver } from './driver';
import type { Book, BookFormat, NewBook, OriginalFormat } from '../core/types';

interface BookRow {
  id: string; title: string; author: string | null; format: string; original_format: string | null;
  file_name: string; file_size: number; language: string | null; publisher: string | null;
  drm_identifier: string | null; added_at: number; updated_at: number; deleted_at: number | null; last_read_at: number | null;
}

const rowToBook = (r: BookRow): Book => ({
  id: r.id, title: r.title, author: r.author, format: r.format as BookFormat,
  originalFormat: r.original_format as OriginalFormat | null, fileName: r.file_name, fileSize: r.file_size,
  language: r.language, publisher: r.publisher, drmIdentifier: r.drm_identifier,
  addedAt: r.added_at, updatedAt: r.updated_at, deletedAt: r.deleted_at, lastReadAt: r.last_read_at,
});

export function createBooksRepo(db: DbDriver, newId: () => string) {
  return {
    async createBook(input: NewBook): Promise<Book> {
      const b: Book = {
        ...input,
        id: input.id ?? newId(),
        author: input.author ?? null,
        language: input.language ?? null,
        publisher: input.publisher ?? null,
        drmIdentifier: input.drmIdentifier ?? null,
        originalFormat: input.originalFormat ?? null,
        deletedAt: null,
        lastReadAt: null,
      };
      await db.run(
        `INSERT INTO books (id,title,author,format,original_format,file_name,file_size,language,publisher,drm_identifier,added_at,updated_at,deleted_at,last_read_at)
         VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
        [b.id, b.title, b.author, b.format, b.originalFormat, b.fileName, b.fileSize, b.language, b.publisher, b.drmIdentifier, b.addedAt, b.updatedAt, b.deletedAt, b.lastReadAt],
      );
      return b;
    },
    async getBook(id: string): Promise<Book | null> {
      const row = await db.get<BookRow>('SELECT * FROM books WHERE id = ?', [id]);
      return row ? rowToBook(row) : null;
    },
    async listBooks(opts: { sort: 'recent' | 'added' | 'title'; query?: string }): Promise<Book[]> {
      const order = {
        recent: 'ORDER BY COALESCE(last_read_at, added_at) DESC, title COLLATE NOCASE ASC',
        added: 'ORDER BY added_at DESC',
        title: 'ORDER BY title COLLATE NOCASE ASC',
      }[opts.sort];
      const where = ['deleted_at IS NULL'];
      const params: unknown[] = [];
      if (opts.query) {
        where.push('(title LIKE ? OR author LIKE ?)');
        const like = `%${opts.query}%`;
        params.push(like, like);
      }
      const rows = await db.all<BookRow>(`SELECT * FROM books WHERE ${where.join(' AND ')} ${order}`, params);
      return rows.map(rowToBook);
    },
    async markDeleted(id: string, at: number): Promise<void> {
      await db.run('UPDATE books SET deleted_at = ?, updated_at = ? WHERE id = ?', [at, at, id]);
    },
    async purgeDeletedBefore(at: number): Promise<number> {
      const r = await db.run('DELETE FROM books WHERE deleted_at IS NOT NULL AND deleted_at <= ?', [at]);
      return r.changes ?? 0;
    },
    async touchLastRead(id: string, at: number): Promise<void> {
      await db.run('UPDATE books SET last_read_at = ? WHERE id = ?', [at, id]);
    },
    async countAll(): Promise<number> {
      const row = await db.get<{ n: number }>('SELECT COUNT(*) AS n FROM books WHERE deleted_at IS NULL');
      return row?.n ?? 0;
    },
  };
}
```

```ts
import { File, Directory, Paths } from 'expo-file-system';
import { readAsStringAsync, EncodingType, getFreeDiskStorageAsync } from 'expo-file-system/legacy';
import { decode } from 'base64-arraybuffer';
import type { FsAdapter } from './fsAdapter';

export function createExpoFsAdapter(): FsAdapter {
  return {
    async freeDiskBytes() { return getFreeDiskStorageAsync(); },
    async ensureDir(path) { new Directory(path).create({ idempotent: true, intermediates: true }); },
    async copy(from, to) { new File(from).copy(to); },
    async exists(path) { return new File(path).exists; },
    async remove(path) { const f = new File(path); if (f.exists) f.delete(); },
    async fileSize(path) { return new File(path).size; },
    async writeBase64(path, b64) { new File(path).write(new Uint8Array(decode(b64))); },
    async readBase64Chunks(path, chunkSize, onChunk) {
      const size = new File(path).size;
      for (let pos = 0; pos < size; pos += chunkSize) {
        const length = Math.min(chunkSize, size - pos);
        const b64 = await readAsStringAsync(path, { position: pos, length, encoding: EncodingType.Base64 });
        await onChunk(b64);
      }
    },
    async directorySize(path) {
      const dir = new Directory(path);
      if (!dir.exists) return 0;
      let total = 0;
      for (const name of dir.list()) total += new File(`${path}/${name}`).size;
      return total;
    },
  };
}
export const documentRoot = () => Paths.document.uri;
```

```ts
import { PROTOCOL_VERSION, type AppToEngine, type EngineToApp } from '../core/bridge/protocol';

declare const __LITERATA_B64__: string;

function send(msg: EngineToApp): void {
  // @ts-expect-error RN webview injection
  const w = window.ReactNativeWebView;
  const payload = JSON.stringify(msg);
  if (w) w.postMessage(payload);
  else console.log('[engine→app]', payload);
}

type Handler = (msg: AppToEngine) => void;
const handlers = new Set<Handler>();

export function onAppMessage(h: Handler): () => void {
  handlers.add(h);
  return () => handlers.delete(h);
}

function receive(raw: unknown): void {
  const data = typeof raw === 'string' ? raw : (raw as { data?: unknown })?.data;
  if (typeof data !== 'string') return;
  try {
    const msg = JSON.parse(data) as AppToEngine;
    for (const h of handlers) h(msg);
  } catch {
    console.log('[engine] unparseable message');
  }
}

// react-native-webview delivers RN→Web posts on `document` (Android) and `window` (iOS)
document.addEventListener('message', receive as EventListener);
window.addEventListener('message', receive as EventListener);

send({ type: 'engineReady', protocolVersion: PROTOCOL_VERSION });
```

```ts
export interface AssemblerState {
  received: number;
  push(seq: number, base64: string): { complete: boolean; bytes: Uint8Array | null };
}

export function createChunkAssembler(totalSize: number, expectedChunkCount = Number.POSITIVE_INFINITY): AssemblerState {
  const parts = new Map<number, Uint8Array>();
  let received = 0;
  return {
    get received() { return received; },
    push(seq, base64) {
      if (seq !== parts.size && !parts.has(seq)) throw new Error(`out-of-order chunk: expected seq ${parts.size}, got ${seq}`);
      const bin = atob(base64);
      const bytes = new Uint8Array(bin.length);
      for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
      if (received + bytes.length > totalSize) throw new Error(`chunk exceeds declared size ${totalSize}`);
      parts.set(seq, bytes);
      received += bytes.length;
      if (received === totalSize) {
        const all = new Uint8Array(totalSize);
        let off = 0;
        for (let s = 0; s < parts.size; s++) { const p = parts.get(s)!; all.set(p, off); off += p.length; }
        return { complete: true, bytes: all };
      }
      return { complete: false, bytes: null };
    },
  };
}
```

```ts
async function coverToJpegBase64(): Promise<string | null> {
  try {
    const coverUrl = await (view as unknown as { getCover?(): Promise<string | null> }).getCover?.();

    const img = new Image();
    await new Promise((res, rej) => { img.onload = res; img.onerror = rej; img.src = coverUrl; });
    const canvas = document.createElement('canvas');

    canvas.width = Math.round(img.width * scale);

    canvas.getContext('2d')!.drawImage(img, 0, 0, canvas.width, canvas.height);
    const dataUrl = canvas.toDataURL('image/jpeg', 0.85);
    return dataUrl.slice(dataUrl.indexOf(',') + 1);
  } catch {
    return null;
  }
}

async function btoaBytes(bytes: Uint8Array): Promise<string> {
  let out = '';

    out += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(out);
}
```

```ts
import { parseMessage } from '../bridge/codec';
import { PROTOCOL_VERSION, type AppToEngine, type EngineMetadataPayload, type EngineToApp } from '../bridge/protocol';
import type { BookFormat, ReaderSettings, TocItem } from '../types';
import type { BookTransport } from './bookTransport';

type Send = (raw: string) => void;

export class ReaderBridge {
  private readyPromise: Promise<number>;
  private resolveReady!: (v: number) => void;
  private progressCbs = new Set<(p: { cfi: string | null; fraction: number; sectionIndex: number }) => void>();
  private errorCbs = new Set<(e: { code: string; message: string }) => void>();
  private bookLoadedWaiter: { resolve: (v: { toc: TocItem[]; metadata: EngineMetadataPayload }) => void; reject: (e: Error) => void } | null = null;
  private ackWaiters: ((seq: number) => void)[] = [];
  private lastAck = -1;
  private bookmarkWaiter: { resolve: (v: { target: string; fraction: number; excerpt: string }) => void; timer: ReturnType<typeof setTimeout> } | null = null;

  constructor(private send: Send, private transport: BookTransport) {
    this.readyPromise = new Promise((resolve) => { this.resolveReady = resolve; });
  }

  whenReady(): Promise<number> { return this.readyPromise; }

  handleRaw(raw: string): void {
    const msg = parseMessage(raw);
    if (msg.type === 'unknown') return;
    this.handle(msg);
  }

  private handle(msg: EngineToApp): void {
    switch (msg.type) {
      case 'engineReady':
        this.resolveReady(msg.protocolVersion);
        break;
      case 'dataAck': {
        this.lastAck = msg.seq;
        const w = this.ackWaiters.shift();
        w?.(msg.seq);
        break;
      }
      case 'bookLoaded':
        this.bookLoadedWaiter?.resolve({ toc: msg.toc, metadata: msg.metadata });
        this.bookLoadedWaiter = null;
        break;
      case 'progress':
        for (const cb of this.progressCbs) cb({ cfi: msg.cfi, fraction: msg.fraction, sectionIndex: msg.sectionIndex });
        break;
      case 'bookmarkCaptured':
        if (this.bookmarkWaiter) {
          clearTimeout(this.bookmarkWaiter.timer);
          this.bookmarkWaiter.resolve({ target: msg.target, fraction: msg.fraction, excerpt: msg.excerpt });
          this.bookmarkWaiter = null;
        }
        break;
      case 'engineError':
        for (const cb of this.errorCbs) cb({ code: msg.code, message: msg.message });
        this.bookLoadedWaiter?.reject(new Error(`${msg.code}: ${msg.message}`));
        this.bookLoadedWaiter = null;
        break;
      default:
        break;
    }
  }

  private emit(msg: AppToEngine): void { this.send(JSON.stringify(msg)); }

  private waitForAck(seq: number): Promise<void> {
    if (this.lastAck >= seq) return Promise.resolve();
    return new Promise((resolve) => this.ackWaiters.push(() => resolve()));
  }

  async openBook(args: { format: BookFormat; path: string; size: number }): Promise<{ toc: TocItem[]; metadata: EngineMetadataPayload }> {
    this.lastAck = -1;
    this.emit({ type: 'openBook', format: args.format, size: args.size });
    const done = new Promise<{ toc: TocItem[]; metadata: EngineMetadataPayload }>((resolve, reject) => {
      this.bookLoadedWaiter = { resolve, reject };
    });
    await this.transport.read(
      { path: args.path, size: args.size },
      async (base64, seq) => {
        this.emit({ type: 'data', seq, base64 });
        await this.waitForAck(seq);
      },
    );
    this.emit({ type: 'dataEnd' });
    return done;
  }

  goTo(target: string): void { this.emit({ type: 'goTo', target }); }
  goToFraction(fraction: number): void { this.emit({ type: 'goToFraction', fraction }); }
  next(): void { this.emit({ type: 'next' }); }
  prev(): void { this.emit({ type: 'prev' }); }
  applySettings(settings: ReaderSettings): void { this.emit({ type: 'applySettings', settings }); }

  captureBookmark(): Promise<{ target: string; fraction: number; excerpt: string }> {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => reject(new Error('captureBookmark timeout')), 5000);
      this.bookmarkWaiter = { resolve, timer };
      this.emit({ type: 'captureBookmark' });
    });
  }

  onProgress(cb: (p: { cfi: string | null; fraction: number; sectionIndex: number }) => void): () => void {
    this.progressCbs.add(cb);
    return () => this.progressCbs.delete(cb);
  }
  onEngineError(cb: (e: { code: string; message: string }) => void): () => void {
    this.errorCbs.add(cb);
    return () => this.errorCbs.delete(cb);
  }
}
```

```ts
export type ImportReason = 'unsupported' | 'no_space' | 'parse_failed' | 'io_error';
export type ImportOutcome = { uri: string; ok: true; bookId: string } | { uri: string; ok: false; reason: ImportReason };

export interface ImportDeps {
  fs: FsAdapter;
  paths: StoragePaths;
  books: BooksRepo;
  extractor: MetadataPort;
  newId?: () => string;
}

const STORED_EXT: Record<Exclude<OriginalFormat, null>, BookFormat> = {
  epub: 'epub', mobi: 'mobi', azw3: 'azw3', cbz: 'cbz', cbr: 'cbz',

export async function runImport(uris: string[], deps: ImportDeps): Promise<ImportOutcome[]> {
  const outcomes: ImportOutcome[] = [];
  await deps.paths.ensureAll();
  for (const uri of uris) {
    try {
      outcomes.push(await importOne(uri, deps));
    } catch {
      outcomes.push({ uri, ok: false, reason: 'io_error' });
    }

  return outcomes;

async function importOne(uri: string, deps: ImportDeps): Promise<ImportOutcome> {
  const original = extFromUri(uri);
  if (!original) return { uri, ok: false, reason: 'unsupported' };

  const size = await deps.fs.fileSize(uri);
  if (!(await hasEnoughSpace(deps.fs, size))) return { uri, ok: false, reason: 'no_space' };

  const id = deps.newId?.() ?? uuidV7();
  const tmpPath = `${deps.paths.booksDir}/${id}.importing`;
  try {
    await deps.fs.copy(uri, tmpPath);
    const extracted = await deps.extractor.extract({ path: tmpPath, format: original, size });
    const format = STORED_EXT[original];
    const finalPath = deps.paths.bookPath(id, format);
    if (extracted.cbzBase64) {
      await deps.fs.writeBase64(finalPath, extracted.cbzBase64);
      await deps.fs.remove(tmpPath);
    } else {
      await deps.fs.copy(tmpPath, finalPath);
      await deps.fs.remove(tmpPath);
    }
    if (extracted.coverJpegBase64) await deps.fs.writeBase64(deps.paths.coverPath(id), extracted.coverJpegBase64);

    const book = await deps.books.createBook({
      id,
      title: extracted.metadata.title ?? uri.split('/').pop() ?? 'Untitled',
      author: extracted.metadata.author,
      format,
      originalFormat: original === 'cbr' ? 'cbr' : null,
      fileName: `${id}.${format}`,
      fileSize: size,
      language: extracted.metadata.language,
      publisher: extracted.metadata.publisher,
      drmIdentifier: extracted.metadata.identifier,
      addedAt: Date.now(),
      updatedAt: Date.now(),
    });
    return { uri, ok: true, bookId: book.id };
  } catch (err) {
    await deps.fs.remove(tmpPath).catch(() => {});
    await deps.fs.remove(deps.paths.bookPath(id, STORED_EXT[original])).catch(() => {});
    await deps.fs.remove(deps.paths.coverPath(id)).catch(() => {});
    if (err instanceof ExtractionError) return { uri, ok: false, reason: 'parse_failed' };
    throw err;
  }
}
```

```ts
import { useCallback, useEffect, useRef, useState } from 'react';
import { useServices } from '../../core/services';
import { ReaderBridge } from '../../core/reader/bridge';
import { createBase64BookTransport } from '../../core/reader/bookTransport';
import { useReaderSettingsStore } from '../settings/readerSettingsStore';
import type { BookFormat, ReaderSettings, TocItem } from '../../core/types';

export interface BridgeLike {
  openBook(a: { format: BookFormat; path: string; size: number }): Promise<{ toc: TocItem[]; metadata: unknown }>;
  goTo(t: string): void; goToFraction(f: number): void; next(): void; prev(): void;
  applySettings(s: ReaderSettings): void;
  onProgress(cb: (p: { cfi: string | null; fraction: number; sectionIndex: number }) => void): () => void;
  onEngineError(cb: (e: { code: string; message: string }) => void): () => void;
  whenReady(): Promise<number>;
}

export interface ReaderSessionDeps {
  getBook(id: string): Promise<{ title: string; format: BookFormat; fileName: string; fileSize: number } | null>;
  getProgress(bookId: string): Promise<{ cfi: string | null; fraction: number } | null>;
  fileExists(fileName: string): Promise<boolean>;
  bookPath(fileName: string): string;
  saveProgress(p: { cfi: string | null; fraction: number; sectionIndex: number | null }): Promise<void>;
  touchLastRead(): Promise<void>;
  makeBridge(): BridgeLike;
}

export function createDefaultDeps(bookId: string): ReaderSessionDeps {
  const services = useServices();
  const settings = useReaderSettingsStore.getState().settings;
  return {
    getBook: async (id) => {
      const b = await services.books.getBook(id);
      return b ? { title: b.title, format: b.format, fileName: b.fileName, fileSize: b.fileSize } : null;
    },
    getProgress: async (id) => {
      const p = await services.progress.getProgress(id);
      return p ? { cfi: p.cfi, fraction: p.fraction } : null;
    },
    fileExists: (fileName) => services.fs.exists(`${services.paths.booksDir}/${fileName}`),
    bookPath: (fileName) => `${services.paths.booksDir}/${fileName}`,
    saveProgress: (p) => services.progress.saveProgress({ bookId, ...p, updatedAt: Date.now() }),
    touchLastRead: () => services.books.touchLastRead(bookId, Date.now()),
    makeBridge: () => new ReaderBridge(bridgeTarget, createBase64BookTransport(services.fs)),
  };
}

// sender target swapped to the real WebView once mounted (see EngineWebView)
export const bridgeTarget: { current: (raw: string) => void } = {
  current: () => { /* dropped pre-mount; engine buffers nothing and we only send after engineReady */ },
};

export function useReaderSession(bookId: string, depsOverride?: ReaderSessionDeps) {
  const [state, setState] = useState<'loading' | 'ready' | 'error'>('loading');
  const [errorKind, setErrorKind] = useState<'missing' | 'open_failed'>('open_failed');
  const [toc, setToc] = useState<TocItem[]>([]);
  const [fraction, setFraction] = useState(0);
  const bridgeRef = useRef<BridgeLike | null>(null);
  const [nonce, setNonce] = useState(0);
  const retry = useCallback(() => setNonce((n) => n + 1), []);

  useEffect(() => {
    let cancelled = false;
    const deps = depsOverride ?? createDefaultDeps(bookId);
    let offProgress: (() => void) | undefined;
    let lastSave = 0;
    void (async () => {
      const book = await deps.getBook(bookId);
      if (!book || !(await deps.fileExists(book.fileName))) {
        if (!cancelled) { setErrorKind('missing'); setState('error'); }
        return;
      }
      const bridge = deps.makeBridge();
      bridgeRef.current = bridge;
      offProgress = bridge.onProgress((p) => {
        setFraction(p.fraction);
        if (Date.now() - lastSave > 1000) { lastSave = Date.now(); void deps.saveProgress(p); void deps.touchLastRead(); }
      });
      bridge.onEngineError(() => { if (!cancelled) { setErrorKind('open_failed'); setState('error'); } });
      try {
        const loaded = await bridge.openBook({ format: book.format, path: deps.bookPath(book.fileName), size: book.fileSize });
        bridge.applySettings(useReaderSettingsStore.getState().settings);
        const prog = await deps.getProgress(bookId);
        if (prog?.cfi) bridge.goTo(prog.cfi);
        else if (prog) bridge.goToFraction(prog.fraction);
        if (!cancelled) { setToc(loaded.toc); setState('ready'); }
      } catch {
        if (!cancelled) { setErrorKind('open_failed'); setState('error'); }
      }
    })();
    return () => { cancelled = true; offProgress?.(); };
  }, [bookId, nonce, depsOverride]);

  return { state, errorKind, toc, fraction, bridge: bridgeRef.current, retry };
}
```

```tsx
import React, { useEffect, useRef } from 'react';
import { StyleSheet, View } from 'react-native';
import { WebView, type WebViewMessageEvent, type WebViewRenderProcessGoneEvent } from 'react-native-webview';
import { buildEngineHtml } from '../../engine/engineHtml';
import { bridgeTarget } from './useReaderSession';
import type { ReaderBridge } from '../../core/reader/bridge';

export function EngineWebView({ bridge, onFatal }: { bridge: ReaderBridge; onFatal(): void }) {
  const ref = useRef<WebView>(null);
  const reloads = useRef(0);

  useEffect(() => {
    bridgeTarget.current = (raw: string) => ref.current?.postMessage(raw);
    return () => { bridgeTarget.current = () => {}; };
  }, []);

  const onMessage = (e: WebViewMessageEvent) => bridge.handleRaw(e.nativeEvent.data);
  const onGone = (_e: WebViewRenderProcessGoneEvent) => {
    if (reloads.current < 1) { reloads.current += 1; ref.current?.reload(); }
    else onFatal();
  };

  return (
    <View style={StyleSheet.absoluteFill}>
      <WebView
        ref={ref}
        source={{ html: buildEngineHtml() }}
        originWhitelist={['*']}
        javaScriptEnabled
        setSupportMultipleWindows={false}
        onMessage={onMessage}
        onRenderProcessGone={onGone}
        onError={onGone}
        style={{ backgroundColor: 'transparent' }}
      />
    </View>
  );
```

```tsx
    onMoveShouldSetPanResponder: () => true,
    onPanResponderGrant: (e) => onScrub(clamp(e.nativeEvent.locationX / (width.current || 1))),
    onPanResponderMove: (e) => onScrub(clamp(e.nativeEvent.locationX / (width.current || 1))),
  })).current;
  return (
    <View style={s.track} onLayout={(e) => { width.current = e.nativeEvent.layout.width; }} {...pan.panHandlers}>
      <View style={[s.fill, { flex: Math.max(fraction, 0.001) }]} />
      <View style={{ flex: Math.max(1 - fraction, 0.001) }} />
    </View>
  );
}

const s = StyleSheet.create({
  track: { flexDirection: 'row', height: 24, borderRadius: 4, overflow: 'hidden', backgroundColor: '#DDD' },

});
```
